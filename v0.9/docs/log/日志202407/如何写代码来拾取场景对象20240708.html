如何写代码来拾取场景对象20240708

viewerPick

调用方法如下：

g_app.xe2ProjectManager.viewerPick(g_app.czmViewer, [0, 0])

在控制台的实现效果如图所示：

事件监听

// 获取点击事件，这是m3t自带的，也可以直接用div.onclick等方法
var ce = g_app.czmViewer.interaction.pointerClick.clickEvent;

// 监听点击事件，执行拾取操作，并打印拾取结果
var d = ce.disposableOn(async e => {
    var result = await g_app.xe2ProjectManager.viewerPick(g_app.czmViewer, [e.offsetX, e.offsetY]);
    console.log(result);
});

// d() 用来取消监听！

拾取结果信息筛选

// 该函数，可以用来获得拾取结果中的Cesium拾取信息！
function getCzmPickedInfo(pickedInfo) {
    let currentPickedInfo = pickedInfo;
    let czmPickResult = undefined;
    do {
        if ('czmPickResult' in currentPickedInfo) {
            czmPickResult = currentPickedInfo['czmPickResult'];
            break;
        }
        currentPickedInfo = currentPickedInfo.childPickedInfo;
    } while (!!currentPickedInfo);
    
    return czmPickResult;
}

// 使用方法举例

// 获取点击事件，这是m3t自带的，也可以直接用div.onclick等方法
var ce = g_app.czmViewer.interaction.pointerClick.clickEvent;

var d = ce.disposableOn(async e => {
    var result = await g_app.xe2ProjectManager.viewerPick(g_app.czmViewer, [e.offsetX, e.offsetY]);
    console.log(`拾取结果`);
    console.log(result);
    if (!result) return;
    const czmPickedInfo = getCzmPickedInfo(result);
    console.log(`Cesium拾取结果`);
    console.log(czmPickedInfo);
});

// d() // 取消鼠标事件监听！

附着信息(AttachedInfo)获取

附着信息的目的是为了辨别拾取结果是哪个拾取操作所为，并附带一些额外使用的信息。例如用户可能在不同的地方多次调用viewerPick，那么如何辨识最后的拾取结果出自指定的viewerPick呢？此时就可以使用AttachedInfo来操作。具体使用方法如下： 可以通过viewerPick的第三个参数，增加拾取时的附着信息

var attachedInfo = { type: '自定义', data: '任意数据！' };
g_app.xe2ProjectManager.viewerPick(g_app.czmViewer, [e.offsetX, e.offsetY], attachedInfo);

可以通过PickedInfo.getFinalAttachedInfo来获取附着信息

const ai = XE2['xe2-base-objects'].PickedInfo.getFinalAttachedInfo(result);

完整示例代码

// 获取点击事件，这是m3t自带的，也可以直接用div.onclick等方法
var ce = g_app.czmViewer.interaction.pointerClick.clickEvent;

var d = ce.disposableOn(async e => {
    var attachedInfo = { type: '自定义', data: '任意数据！' };
    // 1 viewerPick执行时作为第三个参数附带attachedInfo
    var result = await g_app.xe2ProjectManager.viewerPick(g_app.czmViewer, [e.offsetX, e.offsetY], attachedInfo);
    console.log(`拾取结果`);
    console.log(result);
    if (!result) return;
    
    // 2 通过getFinalAttachedInfo从拾取结果中获取附着信息
    const ai = XE2['xe2-base-objects'].PickedInfo.getFinalAttachedInfo(result);
    console.log(`打印附着信息`);
    console.log(ai);
});

// d() // 取消鼠标事件监听！
